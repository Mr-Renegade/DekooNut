cmake_minimum_required(VERSION 3.22)
project(oot_mesh_xlate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LICHII_SOURCE "${REPO}/refs/lichii-mesh" CACHE PATH "Pinned Lichii checkout")
set(BOOST_HEADERS "${REPO}/build/mesh-deps/boost_1_89_0" CACHE PATH "Boost headers")
set(SPIRV_HEADERS "${REPO}/refs/mesh-spirv/include" CACHE PATH "SPIR-V include directory")
add_subdirectory("${REPO}/refs/mesh-fmt" "${CMAKE_BINARY_DIR}/fmt" EXCLUDE_FROM_ALL)
file(GLOB_RECURSE SIRIT_SRC "${REPO}/refs/mesh-sirit/src/*.cpp")
add_library(mesh_sirit STATIC ${SIRIT_SRC})
target_include_directories(mesh_sirit PUBLIC "${REPO}/refs/mesh-sirit/include" "${SPIRV_HEADERS}" PRIVATE "${REPO}/refs/mesh-sirit/src")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(mesh_sirit PRIVATE -include cstdlib)
endif()
file(GLOB_RECURSE RECOMP_SRC "${LICHII_SOURCE}/src/shader_recompiler/*.cpp")
list(FILTER RECOMP_SRC EXCLUDE REGEX "/backend/(glasm|glsl)/")
# patches/*.patch (diff -ru against the pinned checkout's src/): the files they touch are copied into the build tree,
# patched there and compiled in place of the originals; the pinned checkout stays untouched.
# mesh-output-topology.patch: mesh stages that output points or lines (NVN 2 control +0x5f = 1 or 2).
file(GLOB MESH_PATCHES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/patches/*.patch")
set(PATCHED_SOURCE "${CMAKE_BINARY_DIR}/patched")
file(REMOVE_RECURSE "${PATCHED_SOURCE}")
foreach(patch_file ${MESH_PATCHES})
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${patch_file}")
  file(STRINGS "${patch_file}" patch_heads REGEX "^\\+\\+\\+ ")
  foreach(head ${patch_heads})
    string(REGEX REPLACE "^\\+\\+\\+ [^/]+/([^\t ]+).*$" "\\1" rel "${head}")
    if(NOT EXISTS "${PATCHED_SOURCE}/${rel}")
      get_filename_component(rel_dir "${rel}" DIRECTORY)
      file(COPY "${LICHII_SOURCE}/src/${rel}" DESTINATION "${PATCHED_SOURCE}/${rel_dir}")
      list(REMOVE_ITEM RECOMP_SRC "${LICHII_SOURCE}/src/${rel}")
      list(APPEND RECOMP_SRC "${PATCHED_SOURCE}/${rel}")
    endif()
  endforeach()
  # git apply, not patch: git is there wherever the sources were fetched (Windows has no patch program). It works on
  # plain files; --ignore-whitespace takes a checkout with other line ends.
  execute_process(COMMAND git apply -p1 --ignore-whitespace "${patch_file}" WORKING_DIRECTORY "${PATCHED_SOURCE}" RESULT_VARIABLE patch_rc)
  if(NOT patch_rc EQUAL 0)
    message(FATAL_ERROR "${patch_file} does not apply to ${LICHII_SOURCE}")
  endif()
endforeach()
add_library(mesh_recompiler STATIC ${RECOMP_SRC} common_adapter.cpp)
# The runtime's Ampere headers come first: they carry fixes the pinned checkout lacks (ISBEWR high registers among them).
# Only those: runtime/third_party/eglantine/common is a stub that would shadow the real common/ headers.
set(AMPERE_OVERLAY "${CMAKE_BINARY_DIR}/overlay/shader_recompiler/frontend")
file(MAKE_DIRECTORY "${AMPERE_OVERLAY}")
file(GLOB AMPERE_HEADERS RELATIVE "${REPO}/runtime/third_party/eglantine/shader_recompiler/frontend/ampere"
     "${REPO}/runtime/third_party/eglantine/shader_recompiler/frontend/ampere/*.h")
file(MAKE_DIRECTORY "${AMPERE_OVERLAY}/ampere")
foreach(header ${AMPERE_HEADERS})
  file(CREATE_LINK "${REPO}/runtime/third_party/eglantine/shader_recompiler/frontend/ampere/${header}" "${AMPERE_OVERLAY}/ampere/${header}" SYMBOLIC)
endforeach()
target_include_directories(mesh_recompiler PUBLIC include "${CMAKE_BINARY_DIR}/overlay" "${LICHII_SOURCE}/src" "${BOOST_HEADERS}")
target_link_libraries(mesh_recompiler PUBLIC mesh_sirit fmt::fmt)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  target_compile_options(mesh_recompiler PRIVATE -fbracket-depth=1024 -Wno-deprecated-declarations)
  target_compile_options(mesh_recompiler PUBLIC -Wno-pragma-clang-attribute)
endif()
add_executable(oot_mesh_xlate main.cpp)
target_link_libraries(oot_mesh_xlate PRIVATE mesh_recompiler)
if(MINGW)
  target_link_options(oot_mesh_xlate PRIVATE -static -static-libgcc -static-libstdc++)
endif()
